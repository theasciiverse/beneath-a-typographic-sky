# Beneath a Steel Sky (CD v0.0372), from the restart point to Foster's
# first entry into the LINC. Run with
#   typosky walk walkthrough/to-linc.txt --until-linc
#
# Follows M. Shaw's walkthrough (Game Bytes Magazine issue 21, 1994,
# www.ibiblio.org/GameBytes/issue21/misc/bass.html), checked against
# "The Complete Beneath A Steel Sky Walkthrough" (FOURTYoz, Hitiek, Red,
# OLiVER, 1994; bearstrong.net/tekst97/data/spill/bass/) where Shaw is
# vague. Object names, dialogue lines and the order of things are the
# ones the game actually shows (found with the hotspots step).
#
# Steps: look/use/take/talk/exit <thing>, use <item> on <thing>,
# choose <part of a line>, waitfor/waitgone <thing>, wait <cycles>,
# settle, save <slot> <name>. See src/walk.rs.

# --- The factory walkway ------------------------------------------------
use rung                          # the loose rung of the ladder: a metal bar
use metal bar on door             # prise it open; the guard follows, gives up
use door
exit exit                         # back into the factory
use stairs                        # down to the factory floor
exit exit                         # east, to the transporter room

# --- Joey and the transporter -------------------------------------------
look @robot_shell                 # "Junk" until looked at
use circuit board on robot shell  # Joey lives again, as a cleaning droid
look transporter                  # so Foster can ask Hobbins about it
use elevator                      # the alarm brings Hobbins out...
exit @right_exit_rm2              # ...and his workshop is empty for a moment
use cupboard door
take wrench
expect item wrench
talk man                          # Hobbins, back from the alarm
choose where am i
choose best way out
choose so how do i get
choose how come your elevator
choose what's wrong with the transporter
choose get on with my inspection
exit exit
waitfor joey
talk joey
choose start that transporter     # Joey gives it a jump-start
click 230 140 left                # out of the transporter's way
waitgone @top_lift                # it lowers a barrel on the elevator...
use hole                          # ...and Foster climbs down the shaft

# --- The furnace: Reich ---------------------------------------------------
look lock
waitfor joey
talk joey
choose can you open that door     # Reich comes in, and dies in the furnace
use corpse                        # his ID card and dark glasses
talk joey                         # wake Joey up
exit exit
save 1 Top level walkway

# --- Lamb's factory: Anita, the wrench in the cogs, Joey's welder shell --
exit @right_exit_s6
exit @factory_entry
talk woman                        # Anita
choose what are you doing
choose ground level
choose security been              # Lamb sends her off and throws Foster out
choose safety inspector
exit @factory_entry               # straight back in
exit @fact1_right_exit
use wrench on cogs                # production stops
choose no technician
choose don't know anything
take wrench jammed
expect item wrench
exit @fact2_exit_left
use wrench on robot               # loosen the welder's shell
waitfor joey
talk joey
choose new shell                  # Joey moves into the welder

# --- The storeroom: putty ----------------------------------------------
exit @fact1_right_exit
look window
waitfor joey
talk joey
choose never mind

talk joey
choose check out the storeroom
waitgone joey
waitfor joey
settle
talk joey
choose find anything useful       # a fuse box for the alarm
talk joey
choose disable the fuse box
waitgone joey
waitfor joey
settle
exit @fact2_store_exit
use gangway
take putty
expect item putty
exit exit                         # Potts searches Foster, keeps the glasses
choose not a criminal

# --- The power station: light bulb, and power for the elevator ----------
exit @fact2_exit_left
exit @factory_exit
exit @left_exit_s7
exit @left_exit_s6
exit @left_exit_s5
use wrench on @left_skull          # loosen both buttons
use wrench on @right_skull
use @left_skull                   # stand by the left one
waitfor joey
settle
talk joey
choose button on the right        # Joey takes the right one...
use @left_skull                   # ...both at once: the pipe bursts, the man leaves
use @power_switch                 # power off
take light bulb
expect item light bulb
use putty on @socket
use @power_switch                 # the putty blows the bars open
use @right_lever                  # both switches down

# --- Down to the second level --------------------------------------------
exit @right_exit_s18
exit @right_exit_s5
exit @right_exit_s6
look cable
waitfor joey
settle
talk joey

click 120 150 left
talk joey

use id card on slot
use elevator

take cable
expect item cable
save 2 Second level, cable taken
